**Goal:** Add Codex as a second provider on the same item model: a `codex app-server` JSON-RPC client, a translator with a recorded fixture, the provider (approvals, modes, models, context, MCP status, MCP OAuth sign-in), provider choice for new chats, and a live smoke run.

**Architecture:** `CodexSession` spawns `codex app-server` in its own process group, speaks newline-delimited JSON-RPC through `RpcPeer`, and feeds notifications to `CodexTranslator`, which emits `SessionItem`s. Server requests for approvals become `approval` items. Facts come from `model/list`, `thread/tokenUsage/updated`, and `mcpServerStatus/list`. The CLI is found on `PATH` or, failing that, inside the newest official Codex VS Code extension, which bundles it.

**Tech Stack:** as before; no new dependencies.

**What was verified before writing this plan** (2026-10-28, `codex-cli 0.165.0` via npx for the schema; the user's bundled `codex-cli 0.164.0-alpha.11.1` for a live turn):
- Framing: one JSON object per line on stdin/stdout, no `jsonrpc` field. Requests `{ id, method, params }`, responses `{ id, result }` or `{ id, error: { code, message } }`, notifications `{ method, params }`, server requests `{ id, method, params }` answered with `{ id, result }`.
- Handshake: `initialize { clientInfo, capabilities: { experimentalApi: true, requestAttestation: false } }`, then the `initialized` notification.
- `thread/start { cwd, approvalPolicy, sandbox, model? }` returns `{ thread: { id }, model }`; `thread/resume` takes the same plus `threadId`. `turn/start { threadId, input, approvalPolicy?, sandboxPolicy?, model? }` returns at once; the turn is reported by notifications. `turn/interrupt { threadId, turnId }`.
- A live edit turn produced, in order: `turn/started`, `item/started|completed` for `userMessage`, `agentMessage` (with `item/agentMessage/delta`), `reasoning` (often with an empty `summary`), `commandExecution` (with `commandActions`, `aggregatedOutput`, `exitCode`), `thread/tokenUsage/updated` (with `modelContextWindow`), and `turn/completed { turn: { status } }`. It also sent `account/rateLimits/updated`, `turn/moderationMetadata`, and `hook/*`, which AgentDeck ignores.
- Commands arrive wrapped, e.g. `/bin/zsh -lc "grep -n -F 'hello world' notes.md"`.
- `mcpServerStatus/list` returns `{ data: [{ name, runtimeStatus, authStatus, tools, toolsError }] }`; `runtimeStatus` can be `null` (then `mcpServer/startupStatus/updated` notifications carry `starting|ready|failed|cancelled`). `mcpServer/oauth/login { name, threadId }` returns `{ authorizationUrl }`: Codex completes the OAuth itself, so AgentDeck only opens the link.
- `codex login status` prints `Logged in using ChatGPT` and exits 0 when signed in.
- Raw frames contain the user's email, account ids, and hook paths, so the fixture recorder keeps only item, turn, and token-usage notifications and replaces the working directory with `/work`.

```ts
import { type ChildProcess, type SpawnOptions, spawn } from 'node:child_process'

/** Spawns a CLI as the leader of its own process group, so stopping it also stops what it started. */
export function spawnGroup(command: string, args: string[], options: SpawnOptions): ChildProcess {
  return spawn(command, args, { ...options, detached: process.platform !== 'win32' })
}

/** SIGTERM to the process group, then SIGKILL after the grace period. Windows uses taskkill on the tree. */
export async function stopGroup(child: ChildProcess, graceMs = 2000): Promise<void> {
  const pid = child.pid
  if (pid === undefined || child.exitCode !== null || child.signalCode !== null) return
  const exited = new Promise<void>((resolve) => child.once('exit', () => resolve()))
  if (process.platform === 'win32') {
    spawn('taskkill', ['/pid', String(pid), '/T', '/F'], { stdio: 'ignore' })
    await exited
    return
  }
  signal(-pid, 'SIGTERM')
  const timer = setTimeout(() => signal(-pid, 'SIGKILL'), graceMs)
  await exited
  clearTimeout(timer)
}

function signal(pid: number, name: NodeJS.Signals): void {
  try {
    process.kill(pid, name)
  } catch {
    // The group is already gone.
  }
}
```

```ts
import { existsSync, readdirSync } from 'node:fs'
import { homedir } from 'node:os'
import { join } from 'node:path'
import { type Env, findOnPath } from '../../env'

const EXTENSION_ROOTS = ['.vscode/extensions', '.vscode-insiders/extensions', '.cursor/extensions']
const PREFIX = 'openai.chatgpt-'

/** The codex CLI on PATH, else the one bundled with the newest official Codex extension. */
export function findCodex(env: Env, home = env.HOME ?? homedir()): string | null {
  const onPath = findOnPath(env, 'codex')
  if (onPath) return onPath
  const name = process.platform === 'win32' ? 'codex.exe' : 'codex'
  for (const root of EXTENSION_ROOTS) {
    const dir = join(home, root)
    if (!existsSync(dir)) continue
    const extensions = readdirSync(dir)
      .filter((entry) => entry.startsWith(PREFIX))
      .sort((a, b) => compareVersions(b, a))
    for (const extension of extensions) {
      const bin = join(dir, extension, 'bin')
      if (!existsSync(bin)) continue
      for (const platform of readdirSync(bin)) {
        const candidate = join(bin, platform, name)
        if (existsSync(candidate)) return candidate
      }
    }
  }
  return null
}

function compareVersions(a: string, b: string): number {
  const parse = (name: string) => (name.slice(PREFIX.length).match(/^[\d.]+/)?.[0] ?? '').split('.').map(Number)
  const [left, right] = [parse(a), parse(b)]
  for (let i = 0; i < Math.max(left.length, right.length); i++) {
    const difference = (left[i] ?? 0) - (right[i] ?? 0)
    if (difference !== 0) return difference
  }
  return 0
}
```

```ts
import type { Readable, Writable } from 'node:stream'
import { Emitter } from '../../core/emitter'

export type Notification = { method: string; params: unknown }
export type ServerRequest = { id: number | string; method: string; params: unknown }

type Pending = { method: string; resolve: (result: unknown) => void; reject: (error: Error) => void }
type Incoming = {
  id?: number | string
  method?: string
  params?: unknown
  result?: unknown
  error?: { message?: string }
}

/** JSON-RPC as `codex app-server` speaks it: one JSON object per line, no "jsonrpc" field. */
export class RpcPeer {
  private seq = 0
  private buffer = ''
  private readonly pending = new Map<number, Pending>()
  private readonly notifications = new Emitter<Notification>()
  readonly onNotification = this.notifications.event
  private handler: (request: ServerRequest) => Promise<unknown> = async (request) => {
    throw new Error(`Unsupported request: ${request.method}`)
  }

  constructor(
    input: Readable,
    private readonly output: Writable,
    private readonly log: (message: string) => void = () => {}
  ) {
    input.setEncoding('utf8')
    input.on('data', (chunk: string) => this.receive(chunk))
  }

  onRequest(handler: (request: ServerRequest) => Promise<unknown>): void {
    this.handler = handler
  }

  request<T>(method: string, params?: unknown): Promise<T> {
    const id = ++this.seq
    return new Promise<T>((resolve, reject) => {
      this.pending.set(id, { method, resolve: resolve as (result: unknown) => void, reject })
      this.write(params === undefined ? { id, method } : { id, method, params })
    })
  }

  notify(method: string, params?: unknown): void {
    this.write(params === undefined ? { method } : { method, params })
  }

  /** Fails every outstanding request, e.g. when the process exits. */
  close(error: Error): void {
    for (const pending of this.pending.values()) pending.reject(error)
    this.pending.clear()
  }

  private write(message: object): void {
    this.output.write(`${JSON.stringify(message)}\n`)
  }

  private receive(chunk: string): void {
    this.buffer += chunk
    let newline = this.buffer.indexOf('\n')
    while (newline >= 0) {
      const line = this.buffer.slice(0, newline).trim()
      this.buffer = this.buffer.slice(newline + 1)
      if (line) this.dispatch(line)
      newline = this.buffer.indexOf('\n')
    }
  }

  private dispatch(line: string): void {
    let message: Incoming
    try {
      message = JSON.parse(line) as Incoming
    } catch {
      this.log(`Skipped an unparseable app-server line: ${line.slice(0, 200)}`)
      return
    }
    if (message.method !== undefined && message.id !== undefined) {
      void this.serve({ id: message.id, method: message.method, params: message.params })
    } else if (message.method !== undefined) {
      this.notifications.fire({ method: message.method, params: message.params })
    } else {
      this.settle(Number(message.id), message)
    }
  }

  private settle(id: number, message: Incoming): void {
    const pending = this.pending.get(id)
    if (!pending) return
    this.pending.delete(id)
    if (message.error) pending.reject(new Error(`${pending.method}: ${message.error.message ?? 'request failed'}`))
    else pending.resolve(message.result)
  }

  private async serve(request: ServerRequest): Promise<void> {
    try {
      this.write({ id: request.id, result: await this.handler(request) })
    } catch (error) {
      this.write({ id: request.id, error: { code: -32601, message: error instanceof Error ? error.message : String(error) } })
    }
  }
}
```

```ts
const SHELL_WRAPPER = /^\S*\b(?:ba|z)?sh -lc (["'])([\s\S]*)\1$/

/** Codex runs commands as `/bin/zsh -lc "<command>"`; show the command itself. */
export function unwrapShell(command: string): string {
  return command.match(SHELL_WRAPPER)?.[2] ?? command
}

/** The removed and added lines of a unified diff; a diff without hunks is all new content. */
export function splitDiff(diff: string): { oldText: string; newText: string } {
  const lines = diff.split('\n')
  if (!lines.some((line) => line.startsWith('@@'))) return { oldText: '', newText: diff }
  const removed: string[] = []
  const added: string[] = []
  for (const line of lines) {
    if (line.startsWith('---') || line.startsWith('+++')) continue
    if (line.startsWith('-')) removed.push(line.slice(1))
    else if (line.startsWith('+')) added.push(line.slice(1))
  }
  return { oldText: removed.join('\n'), newText: added.join('\n') }
}
```

```ts
import { basename } from 'node:path'
import type { SessionItem, ToolItem, ToolKind } from '../../core/types'
import type { CommandAction, PlanStep, ThreadItem, Turn } from './protocol'
import { splitDiff, unwrapShell } from './text'

/** Tool output kept in the transcript. */
const MAX_OUTPUT = 20_000

const STATUS: Record<string, ToolItem['status']> = {
  inProgress: 'running',
  completed: 'done',
  failed: 'failed',
  declined: 'failed'
}

/** Turns app-server notifications into session items. Facts are the provider's job. */
export class CodexTranslator {
  private readonly items = new Map<string, SessionItem>()
  private readonly types = new Map<string, string>()
  private interruptRequested = false

  constructor(private readonly emit: (item: SessionItem) => void) {}

  interrupted(): void {
    this.interruptRequested = true
  }

  feed(method: string, params: unknown): void {
    switch (method) {
      case 'turn/started': {
        const { turn } = params as { turn: Turn }
        this.interruptRequested = false
        return this.upsert({ kind: 'turn', id: `turn:${turn.id}`, state: 'started' })
      }
      case 'turn/completed':
        return this.turnCompleted((params as { turn: Turn }).turn)
      case 'item/started':
        return this.item((params as { item: ThreadItem }).item, false)
      case 'item/completed':
        return this.item((params as { item: ThreadItem }).item, true)
      case 'item/agentMessage/delta':
      case 'item/reasoning/summaryTextDelta':
      case 'item/commandExecution/outputDelta':
        return this.delta(params as { itemId: string; delta: string })
      case 'turn/plan/updated':
        return this.plan((params as { plan: PlanStep[] }).plan)
    }
  }

  private turnCompleted(turn: Turn): void {
    const id = `turn:${turn.id}`
    const interrupted = turn.status === 'interrupted' || this.interruptRequested
    this.upsert({
      kind: 'turn',
      id,
      state: 'ended',
      stopReason: interrupted ? 'interrupted' : turn.status,
      durationMs: turn.durationMs ?? undefined
    })
    if (turn.status === 'failed' && !interrupted) {
      this.upsert({ kind: 'notice', id: `${id}:error`, level: 'error', text: turn.error?.message ?? 'The turn failed.' })
    }
  }

  private item(item: ThreadItem, completed: boolean): void {
    this.types.set(item.id, item.type)
    switch (item.type) {
      case 'agentMessage':
        return this.upsert({ kind: 'message', id: item.id, role: 'assistant', text: item.text, streaming: !completed })
      case 'reasoning': {
        const current = this.items.get(item.id)
        const text = item.summary.join('\n\n') || (current?.kind === 'thought' ? current.text : '')
        if (text) this.upsert({ kind: 'thought', id: item.id, text, streaming: !completed })
        return
      }
      case 'commandExecution': {
        const command = unwrapShell(item.command)
        const current = this.items.get(item.id)
        const output = item.aggregatedOutput ?? (current?.kind === 'tool' ? current.output : undefined)
        return this.upsert({
          kind: 'tool',
          id: item.id,
          title: `Run ${command}`,
          toolKind: commandKind(item.commandActions),
          status: STATUS[item.status],
          input: { command },
          ...(output ? { output: output.slice(0, MAX_OUTPUT) } : {})
        })
      }
      case 'fileChange':
        this.upsert({
          kind: 'tool',
          id: item.id,
          title: `Edit ${item.changes.map((change) => basename(change.path)).join(', ')}`,
          toolKind: 'edit',
          status: STATUS[item.status],
          paths: item.changes.map((change) => change.path)
        })
        item.changes.forEach((change, index) => {
          const { oldText, newText } = splitDiff(change.diff)
          this.upsert({
            kind: 'diff',
            id: `${item.id}:${index}`,
            path: change.path,
            oldText: change.kind.type === 'add' ? null : oldText,
            newText
          })
        })
        return
      case 'mcpToolCall':
        return this.upsert({
          kind: 'tool',
          id: item.id,
          title: `${item.server} · ${item.tool}`,
          toolKind: 'mcp',
          status: STATUS[item.status],
          input: item.arguments,
          ...(completed ? { output: (item.error?.message ?? mcpText(item.result)).slice(0, MAX_OUTPUT) } : {})
        })
    }
  }

  private delta({ itemId, delta }: { itemId: string; delta: string }): void {
    const current = this.items.get(itemId)
    if (current?.kind === 'message' && current.role === 'assistant') this.upsert({ ...current, text: current.text + delta })
    else if (current?.kind === 'thought') this.upsert({ ...current, text: current.text + delta })
    else if (current?.kind === 'tool') this.upsert({ ...current, output: `${current.output ?? ''}${delta}`.slice(0, MAX_OUTPUT) })
    else if (!current && this.types.get(itemId) === 'reasoning') {
      this.upsert({ kind: 'thought', id: itemId, text: delta, streaming: true })
    }
  }

  private plan(steps: PlanStep[]): void {
    this.upsert({
      kind: 'plan',
      id: 'plan',
      entries: steps.map((step) => ({
        text: step.step,
        status: step.status === 'completed' ? 'done' : step.status === 'inProgress' ? 'in_progress' : 'pending'
      }))
    })
  }

  private upsert(item: SessionItem): void {
    this.items.set(item.id, item)
    this.emit(item)
  }
}

function commandKind(actions: CommandAction[]): ToolKind {
  if (actions.length > 0 && actions.every((action) => action.type === 'read')) return 'read'
  if (actions.some((action) => action.type === 'search' || action.type === 'listFiles')) return 'search'
  return 'execute'
}

function mcpText(result: { content: unknown[] } | null): string {
  if (!result) return ''
  return result.content
    .map((part) => {
      const block = part as { type?: string; text?: string }
      return block.type === 'text' ? (block.text ?? '') : JSON.stringify(part)
    })
    .join('\n')
}
```

```ts
export const codexProvider: ProviderDefinition = {

  label: 'Codex',

class CodexSession implements ProviderSession {
  nativeId = ''
  private readonly items = new Emitter<SessionItem>()
  private readonly facts = new Emitter<SessionFact>()
  private readonly exits = new Emitter<{ error?: string }>()
  readonly onItem = this.items.event
  readonly onFact = this.facts.event
  readonly onExit = this.exits.event
  private readonly translator = new CodexTranslator((item) => this.items.fire(item))
  private readonly approvals = new Map<string, { item: ApprovalItem; resolve: (decision: string) => void }>()
  private readonly startup = new Map<string, McpServer['status']>()
  private models: { id: string; label: string }[] = []
  private model: string | undefined
  private mode: PermissionMode
  private turnId: string | undefined
  private ready = false
  private closed = false
  private stderr = ''

  static async start(path: string, options: OpenOptions): Promise<CodexSession> {
    const child = spawnGroup(path, ['app-server'], { cwd: options.cwd, env: options.env, stdio: ['pipe', 'pipe', 'pipe'] })
    const session = new CodexSession(child, new RpcPeer(child.stdout!, child.stdin!), options)
    try {
      await session.handshake()
    } catch (error) {
      session.closed = true
      await stopGroup(child)
      throw error
    }
    // Facts are emitted after open() resolves, once the caller has subscribed.
    void session.loadFacts()
    return session
  }

  private constructor(
    private readonly child: ChildProcess,
    private readonly rpc: RpcPeer,
    private readonly options: OpenOptions
  ) {
    this.mode = options.mode
    this.model = options.model
    rpc.onNotification(({ method, params }) => this.notification(method, params))
    rpc.onRequest((request) => this.serverRequest(request))
    child.stderr?.setEncoding('utf8')
    child.stderr?.on('data', (chunk: string) => {
      this.stderr = (this.stderr + chunk).slice(-STDERR_TAIL)
    })
    child.on('exit', (code, signal) => {
      rpc.close(new Error('codex app-server exited'))
      const error = this.closed ? undefined : `codex app-server exited (${code ?? signal}).\n${this.stderr.trim()}`
      this.exits.fire({ error })
    })
  }

  async prompt(parts: PromptPart[]): Promise<void> {
    await this.rpc.request('turn/start', {
      threadId: this.nativeId,
      input: [{ type: 'text', text: encodePrompt(parts, this.options.cwd), text_elements: [] }],
      approvalPolicy: policy(this.mode).approvalPolicy,
      sandboxPolicy: sandboxPolicy(this.mode, this.options.cwd),
      model: this.model ?? null
    })
  }

  async cancel(): Promise<void> {
    if (!this.turnId) return
    this.translator.interrupted()
    await this.rpc.request('turn/interrupt', { threadId: this.nativeId, turnId: this.turnId })
  }

  async setModel(id: string): Promise<void> {
    this.model = id
    this.facts.fire({ kind: 'model', current: id, available: this.models })
  }

  async setMode(mode: PermissionMode): Promise<void> {
    this.mode = mode
    this.facts.fire({ kind: 'mode', current: mode, available: MODE_LIST })
  }

  answer(itemId: string, optionId: string): void {
    const pending = this.approvals.get(itemId)
    if (!pending) return
    this.approvals.delete(itemId)
    this.items.fire({ ...pending.item, resolved: optionId })
    pending.resolve(optionId === 'allow' ? 'accept' : optionId === 'always' ? 'acceptForSession' : 'decline')
  }

  async refreshMcp(): Promise<void> {
    await refreshUntilSettled(
      async () => {
        const { data } = await this.rpc.request<{ data: McpServerStatus[] }>('mcpServerStatus/list', {})
        return data.map((server) => this.mcpServer(server))
      },
      (servers) => this.facts.fire({ kind: 'mcp', servers }),
      { intervalMs: 1000, attempts: 30, closed: () => this.closed }
    )
  }

  async reconnectMcp(): Promise<void> {
    await this.rpc.request('config/mcpServer/reload')
    await this.refreshMcp()
  }

  async mcpSignIn(server: string): Promise<McpSignIn> {
    const { authorizationUrl } = await this.rpc.request<{ authorizationUrl: string }>('mcpServer/oauth/login', {
      name: server,
      threadId: this.nativeId
    })
    return { kind: 'url', url: authorizationUrl }
  }

  async dispose(): Promise<void> {
    this.closed = true
    for (const id of [...this.approvals.keys()]) this.answer(id, 'deny')
    await stopGroup(this.child)
  }

  private async handshake(): Promise<void> {
    await this.rpc.request('initialize', {
      clientInfo: { name: 'agentdeck', title: 'AgentDeck', version: '0.0.1' },
      capabilities: { experimentalApi: true, requestAttestation: false }
    })
    this.rpc.notify('initialized')
    const params = { cwd: this.options.cwd, ...policy(this.mode), model: this.options.model ?? null }
    const result = this.options.resumeId
      ? await this.rpc.request<{ thread: { id: string }; model: string }>('thread/resume', { ...params, threadId: this.options.resumeId })
      : await this.rpc.request<{ thread: { id: string }; model: string }>('thread/start', params)
    this.nativeId = result.thread.id
    this.model ??= result.model
  }

  private async loadFacts(): Promise<void> {
    this.ready = true
    try {
      const { data } = await this.rpc.request<{ data: ModelInfo[] }>('model/list', {})
      this.models = data.filter((model) => !model.hidden).map((model) => ({ id: model.id, label: model.displayName }))
    } catch {
      // The model picker then offers only the current model.
    }
    this.facts.fire({ kind: 'mode', current: this.mode, available: MODE_LIST })
    this.facts.fire({ kind: 'model', current: this.model ?? '', available: this.models })
    await this.refreshMcp().catch(() => undefined)
  }

  private notification(method: string, params: unknown): void {
    this.translator.feed(method, params)
    switch (method) {
      case 'turn/started':
        this.turnId = (params as { turn: { id: string } }).turn.id
        return
      case 'turn/completed':
        this.turnId = undefined
        return
      case 'thread/tokenUsage/updated': {
        const { tokenUsage } = params as { tokenUsage: TokenUsage }
        if (tokenUsage.modelContextWindow) {
          this.facts.fire({ kind: 'context', usedTokens: tokenUsage.last.totalTokens, maxTokens: tokenUsage.modelContextWindow })
        }
        return
      }
      case 'mcpServer/startupStatus/updated': {
        const { name, status } = params as { name: string; status: string }
        this.startup.set(name, STARTUP[status] ?? 'pending')
        if (this.ready) void this.refreshMcp().catch(() => undefined)
        return
      }
    }
  }

  private serverRequest({ method, params }: ServerRequest): Promise<unknown> {
    if (method === 'item/commandExecution/requestApproval') {
      const request = params as { itemId: string; command?: string | null; reason?: string | null }
      const detail = request.command ? unwrapShell(request.command) : (request.reason ?? undefined)
      return this.ask(request.itemId, 'Allow Codex to run this command?', detail)
    }
    if (method === 'item/fileChange/requestApproval') {
      const request = params as { itemId: string; reason?: string | null; grantRoot?: string | null }
      if (this.mode === 'accept-edits') return Promise.resolve({ decision: 'accept' })
      return this.ask(request.itemId, 'Allow Codex to change files?', request.grantRoot ?? request.reason ?? undefined)
    }
    return Promise.reject(new Error(`AgentDeck does not handle ${method} yet.`))
  }

  private ask(itemId: string, title: string, detail: string | undefined): Promise<{ decision: string }> {
    const item: ApprovalItem = {
      kind: 'approval',
      id: `approval:${itemId}`,
      title,
      detail,
      options: [
        { id: 'allow', label: 'Allow once' },
        { id: 'always', label: 'Allow for this session' },
        { id: 'deny', label: 'Deny' }
      ]
    }
    return new Promise((resolve) => {
      this.approvals.set(item.id, { item, resolve: (decision) => resolve({ decision }) })
      this.items.fire(item)
    })
  }

  private mcpServer(server: McpServerStatus): McpServer {
    const status =
      (server.runtimeStatus ? RUNTIME[server.runtimeStatus] : undefined) ??
      this.startup.get(server.name) ??
      (server.authStatus === 'notLoggedIn' ? 'needs-auth' : 'connected')
    return { name: server.name, status, tools: Object.keys(server.tools ?? {}), error: server.toolsError ?? undefined }
  }
}
```

```ts
// Records a real Codex edit turn's app-server notifications as a translator fixture.
// Only item, turn, and token-usage notifications are kept; other frames carry account details.
import { mkdirSync, mkdtempSync, realpathSync, writeFileSync } from 'node:fs'
import { tmpdir } from 'node:os'
import { join } from 'node:path'
import { captureEnv } from '../src/env'
import { spawnGroup, stopGroup } from '../src/process'

const OUT = 'test/fixtures/codex/edit-turn.jsonl'
const KEEP = /^(item\/|turn\/(started|completed|plan\/updated)$|thread\/tokenUsage\/updated$)/

async function main(): Promise<void> {
  const env = await captureEnv()
  const cli = findCodex(env)
  if (!cli) throw new Error('codex CLI not found')
  const cwd = realpathSync(mkdtempSync(join(tmpdir(), 'agentdeck-record-')))
  writeFileSync(join(cwd, 'notes.md'), '# Notes\n\nhello world\n')

  const child = spawnGroup(cli, ['app-server'], { cwd, env, stdio: ['pipe', 'pipe', 'inherit'] })
  const rpc = new RpcPeer(child.stdout!, child.stdin!)
  const lines: string[] = []
  const completed = new Promise<void>((resolve) => {
    rpc.onNotification(({ method, params }) => {
      if (KEEP.test(method)) lines.push(JSON.stringify({ method, params }).replaceAll(cwd, '/work'))
      if (method === 'turn/completed') resolve()
    })
  })
  await rpc.request('initialize', {
    clientInfo: { name: 'agentdeck', title: 'AgentDeck', version: '0.0.1' },
    capabilities: { experimentalApi: true, requestAttestation: false }
  })
  rpc.notify('initialized')
  const { thread } = await rpc.request<{ thread: { id: string } }>('thread/start', {
    cwd,
    approvalPolicy: 'never',
    sandbox: 'workspace-write',
    ephemeral: true
  })
  await rpc.request('turn/start', {
    threadId: thread.id,
    input: [{ type: 'text', text: 'In notes.md, replace "hello world" with "goodbye world" using apply_patch. Then reply with one short sentence.', text_elements: [] }]
  })
  await completed
  await stopGroup(child)
  mkdirSync('test/fixtures/codex', { recursive: true })
  writeFileSync(OUT, `${lines.join('\n')}\n`)
  console.log(`wrote ${lines.length} notifications to ${OUT}`)
}

main().catch((error) => {
  console.error('RECORD FAILED:', error)
  process.exit(1)
})
```
