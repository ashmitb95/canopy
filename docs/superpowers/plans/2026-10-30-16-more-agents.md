**Goal:** Plan 16 adds three agents and two configuration helpers:
- **Cursor, OpenCode, and Gemini CLI** join Grok as ACP agents, each defined by a spec (decision 27).
- **Any other ACP agent** can be added from settings (`agentdeck.customAgents`).
- ***Open Provider Config…*** opens the config files of the installed CLIs and the workspace's `.agentdeck/config.toml`.

**Verified flags (October 2026):**
- **Cursor:** `cursor-agent acp`. The official docs say `agent acp`, but Grok's CLI also installs an `agent` binary in `~/.grok/bin`, so AgentDeck uses Cursor's other name, `cursor-agent`, found in `~/.local/bin`. Sign-in is `cursor-agent login` or `CURSOR_API_KEY`. Cursor also sends the blocking extension methods `cursor/ask_question` and `cursor/create_plan`. They are answered with a JSON-RPC error, which is AgentDeck's existing handling for unknown methods, so the agent never hangs.
- **OpenCode:** `opencode acp`; sign-in is `opencode auth login`.
- **Gemini CLI:** `gemini --acp`. It was stable from 0.50; older builds need `--experimental-acp`. Sign-in is `GEMINI_API_KEY`, or the Google login stored in `~/.gemini/oauth_creds.json`.

None of these CLIs is installed on the development machine, so this plan does not record fixtures for them and does not run live checks. The hand-off tells the user how to verify each agent once it is installed.

```ts
export const cursor: AcpAgentSpec = {
  id: 'cursor',
  label: 'Cursor',
  // Cursor's docs name the binary `agent`; Grok's CLI installs another `agent`, so use Cursor's `cursor-agent`.
  command: 'cursor-agent',
  installUrl: 'https://cursor.com/docs/cli/installation',
  args: ['acp'],
  loginArgs: ['login'],
  installDirs: (home) => [join(home, '.local', 'bin')],
  authMethod: () => undefined,
  // Cursor keeps its login in its own store; without an API key, a missing login surfaces as the agent's own error.
  signedIn: () => true
}

export const opencode: AcpAgentSpec = {
  id: 'opencode',
  label: 'OpenCode',
  command: 'opencode',
  installUrl: 'https://opencode.ai/docs',
  args: ['acp'],
  loginArgs: ['auth', 'login'],
  installDirs: (home) => [join(home, '.opencode', 'bin'), join(home, '.local', 'bin')],
  authMethod: () => undefined,
  signedIn: (_env, home) => existsSync(join(home, '.local', 'share', 'opencode', 'auth.json'))
}

export const gemini: AcpAgentSpec = {
  id: 'gemini',
  label: 'Gemini',
  command: 'gemini',
  installUrl: 'https://geminicli.com/docs/get-started/installation',
  args: ['--acp'],
  // Gemini signs in interactively on first run.
  loginArgs: [],
  installDirs: () => [],
  authMethod: (advertised, env) =>
    env.GEMINI_API_KEY?.trim() && advertised.includes('gemini-api-key')
      ? 'gemini-api-key'
      : advertised.includes('oauth-personal')
        ? 'oauth-personal'
        : undefined,
  signedIn: (env, home) => Boolean(env.GEMINI_API_KEY?.trim()) || existsSync(join(home, '.gemini', 'oauth_creds.json'))
}

const BUILT_IN = new Set(['claude', 'codex', 'grok', 'cursor', 'opencode', 'gemini'])

/** An ACP agent from `agentdeck.customAgents`: `{ id, label, command, args, loginArgs? }`. */
export function customAgent(entry: Record<string, unknown>): AcpAgentSpec | undefined {
  const { id, label, command, args, loginArgs } = entry
  const strings = (value: unknown) => (Array.isArray(value) && value.every((item) => typeof item === 'string') ? (value as string[]) : undefined)
  if (typeof id !== 'string' || !/^[a-z0-9-]+$/.test(id) || BUILT_IN.has(id)) return undefined
  if (typeof label !== 'string' || typeof command !== 'string' || !strings(args)) return undefined
  return {
    id,
    label,
    command,
    installUrl: 'https://agentclientprotocol.com/overview/agents',
    args: strings(args) ?? [],
    loginArgs: strings(loginArgs) ?? [],
    installDirs: () => [],
    authMethod: () => undefined,
    signedIn: () => true
  }
}
```

```ts
import { cursor, gemini, grok, opencode } from './acp/agents'
import { acpProvider } from './acp/provider'
import type { AcpAgentSpec } from './acp/spec'
import { claudeProvider } from './claude/provider'
import { codexProvider } from './codex/provider'

/** Built-in providers, then any ACP agents the user added in settings. */
export function buildProviders(custom: AcpAgentSpec[] = []): ReadonlyMap<string, ProviderDefinition> {
  return new Map(
    [claudeProvider, codexProvider, ...[grok, cursor, opencode, gemini, ...custom].map(acpProvider)].map((provider) => [provider.id, provider])
  )
}

export const providers = buildProviders()
```

`src/ui/config-files.ts`:

```ts
import { existsSync } from 'node:fs'
import { join } from 'node:path'

/** Provider and AgentDeck config files that exist on this machine, for *Open Provider Config…*. */
export function configFiles(home: string, root: string | undefined): { label: string; path: string }[] {
  const candidates = [
    { label: 'Claude settings', path: join(home, '.claude', 'settings.json') },
    { label: 'Codex config', path: join(home, '.codex', 'config.toml') },
    { label: 'Grok config', path: join(home, '.grok', 'config.toml') },
    { label: 'Gemini settings', path: join(home, '.gemini', 'settings.json') },
    { label: 'OpenCode config', path: join(home, '.config', 'opencode', 'opencode.json') },
    ...(root ? [{ label: 'AgentDeck workspace config', path: join(root, '.agentdeck', 'config.toml') }] : [])
  ]
  return candidates.filter((candidate) => existsSync(candidate.path))
}
```
